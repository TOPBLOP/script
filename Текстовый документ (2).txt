cp ~/.config/hypr/hyprland.lua ~/.config/hypr/hyprland.lua.working_bak
cat << 'EOF' > ~/.config/hypr/hyprland.lua
-- МОНИТОР --
hl.monitor({
    output   = "",
    mode     = "preferred",
    position = "auto",
    scale    = 1,
})

-- ПРОГРАММЫ --
local terminal    = "kitty"
local fileManager = "nautilus"
local menu        = "rofi -show drun"
local home        = os.getenv("HOME")

-- АВТОЗАПУСК ПРИ ВКЛЮЧЕНИИ --
hl.on("hyprland.start", function()
    hl.exec_cmd("nm-applet")
    hl.exec_cmd("blueman-applet")
    hl.exec_cmd("waybar")
    hl.exec_cmd("awww-daemon")
    hl.exec_cmd("swaync")
    hl.exec_cmd("hypridle")
    hl.exec_cmd("systemctl --user start hyprpolkitagent")
end)

-- ПЕРЕМЕННЫЕ ОКРУЖЕНИЯ --
hl.env("XCURSOR_SIZE", "24")
hl.env("HYPRCURSOR_SIZE", "24")

-- ВНЕШНИЙ ВИД И БЛЮР --
hl.config({
    general = {
        gaps_in     = 5,
        gaps_out    = 12,
        border_size = 2,
        col = {
            active_border   = { colors = {"rgba(33ccffee)", "rgba(00ff99ee)"}, angle = 45 },
            inactive_border = "rgba(595959aa)",
        },
        resize_on_border = false,
        allow_tearing    = false,
        layout           = "dwindle",
    },

    decoration = {
        rounding         = 12,
        rounding_power   = 2,
        active_opacity   = 0.95,
        inactive_opacity = 0.88,

        shadow = {
            enabled      = true,
            range        = 12,
            render_power = 3,
            color        = 0x66000000,
        },

        blur = {
            enabled  = true,
            size     = 6,
            passes   = 3,
            vibrancy = 0.2,
        },
    },

    animations = {
        enabled = true,
    },

    dwindle = {
        preserve_split = true,
    },

    misc = {
        force_default_wallpaper = 0,
        disable_hyprland_logo   = true,
    },

    input = {
        kb_layout    = "us,ru",
        kb_options   = "grp:alt_shift_toggle",
        follow_mouse = 1,
        sensitivity  = 0,
        touchpad = {
            natural_scroll = true,
        },
    },
})

-- АНИМАЦИИ --
hl.curve("easeOutQuint",   { type = "bezier", points = { {0.23, 1},    {0.32, 1}    } })
hl.curve("almostLinear",   { type = "bezier", points = { {0.5, 0.5},   {0.75, 1}    } })
hl.curve("quick",          { type = "bezier", points = { {0.15, 0},    {0.1, 1}     } })
hl.curve("easy",           { type = "spring", mass = 1, stiffness = 238.1191, damping = 24.21279333 })

hl.animation({ leaf = "global",        enabled = true, speed = 10,   bezier = "default" })
hl.animation({ leaf = "border",        enabled = true, speed = 5.39, bezier = "easeOutQuint" })
hl.animation({ leaf = "windows",       enabled = true, speed = 4.79, spring = "easy" })
hl.animation({ leaf = "windowsIn",     enabled = true, speed = 4.1,  spring = "easy", style = "popin 87%" })
hl.animation({ leaf = "windowsOut",    enabled = true, speed = 1.49, bezier = "quick", style = "popin 87%" })
hl.animation({ leaf = "fade",          enabled = true, speed = 3.03, bezier = "quick" })
hl.animation({ leaf = "workspaces",    enabled = true, speed = 1.94, bezier = "almostLinear", style = "fade" })

-- ГОРЯЧИЕ КЛАВИШИ (ВСЕ ИЗ ТВОЕГО РАЙСА) --
hl.bind("SUPER + Return",       hl.dsp.exec(terminal))
hl.bind("SUPER + SHIFT + Return", hl.dsp.exec("[float; size 800 550] " .. terminal))
hl.bind("SUPER + Q",            hl.dsp.window.kill("active"))
hl.bind("CTRL + ALT + Delete",  hl.dsp.exit())
hl.bind("SUPER + E",            hl.dsp.exec(fileManager))
hl.bind("SUPER + SHIFT + E",    hl.dsp.exec("kitty yazi"))
hl.bind("SUPER + Space",        hl.dsp.window.toggle_floating())
hl.bind("SUPER + D",            hl.dsp.exec(menu))
hl.bind("SUPER + J",            hl.dsp.layout("togglesplit"))
hl.bind("SUPER + R",            hl.dsp.exec(home .. "/.config/hypr/scripts/wbrestart.sh"))
hl.bind("SUPER + B",            hl.dsp.exec('xdg-open "https://"'))
hl.bind("SUPER + L",            hl.dsp.exec(home .. "/.config/hypr/scripts/hyprlock.sh"))
hl.bind("SUPER + SHIFT + F",    hl.dsp.window.fullscreen(0))
hl.bind("SUPER + SHIFT + S",    hl.dsp.exec(home .. "/.config/hypr/scripts/screenshot.sh"))
hl.bind("SUPER + W",            hl.dsp.exec(home .. "/.config/hypr/scripts/wppicker.sh"))
hl.bind("SUPER + SHIFT + Q",    hl.dsp.exec(home .. "/.config/hypr/scripts/KillActiveProcess.sh"))
hl.bind("SUPER + C",            hl.dsp.exec("hyprpicker -a"))
hl.bind("SUPER + CTRL + B",     hl.dsp.exec(home .. "/.config/hypr/scripts/WaybarStyles.sh"))
hl.bind("SUPER + ALT + B",      hl.dsp.exec(home .. "/.config/hypr/scripts/WaybarLayout.sh"))
hl.bind("SUPER + H",            hl.dsp.exec("pkill -SIGUSR1 waybar"))

-- ФОКУС И ПЕРЕМЕЩЕНИЕ ОКОН --
hl.bind("SUPER + left",         hl.dsp.focus("l"))
hl.bind("SUPER + right",        hl.dsp.focus("r"))
hl.bind("SUPER + up",           hl.dsp.focus("u"))
hl.bind("SUPER + down",         hl.dsp.focus("d"))

hl.bind("SUPER + CTRL + left",  hl.dsp.window.move("l"))
hl.bind("SUPER + CTRL + right", hl.dsp.window.move("r"))
hl.bind("SUPER + CTRL + up",    hl.dsp.window.move("u"))
hl.bind("SUPER + CTRL + down",  hl.dsp.window.move("d"))

-- РАБОЧИЕ СТОЛЫ (1-10) --
for i = 1, 10 do
    local key = (i == 10) and "0" or tostring(i)
    hl.bind("SUPER + " .. key,           hl.dsp.workspace(i))
    hl.bind("SUPER + SHIFT + " .. key,   hl.dsp.window.move({ workspace = i }))
end

-- МЫШЬ И ТАЧПАД --
hl.bind("SUPER + mouse_down", hl.dsp.workspace("e+1"))
hl.bind("SUPER + mouse_up",   hl.dsp.workspace("e-1"))
hl.bind("SUPER + mouse:272",  hl.dsp.window.move_mouse(),   { flags = "m" })
hl.bind("SUPER + mouse:273",  hl.dsp.window.resize_mouse(), { flags = "m" })

hl.gesture({ fingers = 3, direction = "horizontal", action = "workspace" })
EOF
hyprctl reload
